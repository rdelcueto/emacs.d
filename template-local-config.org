#+TITLE: Local Machine Configuration Template
#+PROPERTY: header-args:emacs-lisp :mkdirp yes :results silent

* Instructions
Copy or symlink this file to =local-config.org= to customize your local machine configuration.
=local-config.org= is consumed by =README.org= and is not tracked by Git, allowing
host-specific layer selections, paths, and secrets.

* Pluggable Feature Modules
All feature modules are opt-in. Comment out any module below to prevent loading:

#+begin_src emacs-lisp :tangle init.el
  ;; 6. Pluggable Feature Modules (Host Selection)
  (defvar active-modules
    '(mod-git          ; Magit, Diff-hl, Ediff, Ztree
      mod-treesit      ; Treesit-env
      mod-lsp          ; Eglot
      mod-completion   ; Corfu, Cape, Tempel
      mod-citre        ; Citre
      lang-python      ; Python, IPython
      lang-rust        ; Rustic, Cargo
      lang-c-cpp       ; C/C++, CUDA
      lang-clojure     ; Cider
      lang-racket      ; Racket-mode
      lang-lispy       ; Lispy, Lispyville
      mod-org          ; Org-mode, Org-modern, Agenda, Transclusion
      mod-notes        ; Denote, Journal, Consult-notes
      mod-terminal     ; Ghostel
      mod-dired        ; Dired, Dired-subtree, fd-dired
      mod-reading      ; PDF/EPUB, Calibre, Org-noter, Asciidoc
      mod-ai)          ; LLMs (GPTel), MCP
    "List of active feature modules to load on this host.")

  (defvaralias 'rdelcueto/active-modules 'active-modules)

  (dolist (mod active-modules)
    (require mod nil t))

  ;; 7. Host & Environment Overrides
  (require 'local-overrides nil t)
  (let ((local-file (expand-file-name "local.el" user-emacs-directory)))
    (when (file-exists-p local-file)
      (load local-file nil t)))
#+end_src
